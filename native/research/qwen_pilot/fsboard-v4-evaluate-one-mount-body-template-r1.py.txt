def require(condition, message):
    if not condition:
        raise RuntimeError(message)


def file_sha(path):
    require(path.is_file() and not path.is_symlink(), f"Missing regular file: {path.name}")
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def read_pinned(path, digest):
    payload = path.read_bytes()
    require(hashlib.sha256(payload).hexdigest() == digest, f"Payload pin mismatch: {path.name}")
    return payload


def write_json(path, report):
    with path.open("x", encoding="utf-8") as stream:
        json.dump(report, stream, indent=2, allow_nan=False)
        stream.write("\n")


def extract_checked(archive, destination, max_files, max_bytes):
    with tarfile.open(archive, "r:gz") as tar:
        members = tar.getmembers()
        require(len(members) <= max_files and sum(m.size for m in members) <= max_bytes,
                "Unexpected archive size or member count")
        seen = set()
        for member in members:
            relative = PurePosixPath(member.name)
            require(not relative.is_absolute() and ".." not in relative.parts and relative.parts
                    and relative not in seen and (member.isdir() or member.isfile()),
                    "Invalid archive member")
            seen.add(relative)
        for member in members:
            target = destination.joinpath(*PurePosixPath(member.name).parts)
            if member.isdir():
                target.mkdir(parents=True, exist_ok=True)
            else:
                target.parent.mkdir(parents=True, exist_ok=True)
                with tar.extractfile(member) as source, target.open("xb") as output:
                    shutil.copyfileobj(source, output)

def capture_records(local, remote, expected, copied, *, strict=False):
    from fsboard_v4_evaluation_launch import JSON_LIMIT, read_object

    def capture(relative):
        source = local / relative
        if not source.exists() and not source.is_symlink():
            return
        require(source.is_file() and not source.is_symlink() and source.resolve() == source,
                "Unexpected non-regular evaluation result")
        with source.open("rb") as stream:
            payload = stream.read(JSON_LIMIT + 1)
        require(len(payload) <= JSON_LIMIT, "Unexpected evaluation result size")
        if relative == "runtime_receipt.json":
            # The direct writer ends its indented object with a closing brace on its own line.
            if not strict and not payload.endswith((b"\n}\n", b"\n}\r\n")):
                return
            require(payload.endswith(b"\n"), "Incomplete evaluation runtime receipt")
        require(payload, "Unexpected evaluation result size")
        read_object(payload, relative)
        target = remote / relative
        target.parent.mkdir(parents=True, exist_ok=True)
        created = False
        try:
            with target.open("xb") as stream:
                created = True
                require(stream.write(payload) == len(payload), "Incomplete evaluation result write")
            require(target.read_bytes() == payload, "Captured evaluation result readback failed")
        except (Exception, KeyboardInterrupt) as error:
            if created:
                try:
                    target.unlink(missing_ok=True)
                except OSError as cleanup_error:
                    error.add_note(f"Could not remove partial evaluation capture {target}: {cleanup_error}")
            raise
        copied[relative] = {"sha256": hashlib.sha256(payload).hexdigest(), "bytes": len(payload)}

    errors = []
    for relative in sorted(expected):
        if relative in copied:
            continue
        try:
            capture(relative)
        except Exception as error:
            if not strict:
                raise
            errors.append(error)
    if strict:
        for relative, receipt in copied.items():
            target = remote / relative
            try:
                require(target.stat().st_size == receipt["bytes"] and file_sha(target) == receipt["sha256"],
                        "Captured evaluation result changed")
            except Exception as error:
                errors.append(error)
        if errors:
            raise errors[0]


def verify_complete(remote, copied, candidates, clips, plan, controls):
    import fsboard_task_evaluate as evaluation
    from fsboard_v4_evaluation_launch import EXPERIMENT_ID, read_object

    def document(relative):
        require(relative in copied, "Missing completed evaluation result")
        return read_object(read_pinned(remote / relative, copied[relative]["sha256"]), relative)

    expected_inputs = read_object(controls.payload, "attested holdout processor inputs")["examples"]
    summaries = {}
    conditions = evaluation.CONDITIONS
    for candidate, binding in candidates.items():
        binding_name = f"bindings/{candidate}.json"
        require(document(binding_name) == binding, "Evaluation candidate binding mismatch")
        scores = {condition: [] for condition in conditions}
        for clip in clips:
            clip_id = clip["id"]
            for condition in conditions:
                row = document(f"{candidate}/{clip_id}/{condition}.json")
                require(row.get("candidate") == candidate and row.get("condition") == condition
                        and row.get("clip_id") == clip_id and row.get("reference") == clip["reference"]
                        and row.get("signer_id") == clip["signer_id"]
                        and row.get("publisher_split") == clip["split"]
                        and row.get("control") == plan["examples"][clip_id]["conditions"][condition]
                        and row.get("model_input") == expected_inputs[clip_id][condition],
                        "Evaluation row disagrees with the admitted clip, condition or processor attestation")
                nll = row["reference_nll"]
                require(nll["tokens_including_eos"] == row["model_input"]["supervised_tokens_including_eos"],
                        "Incomplete teacher-forced reference loss")
                scores[condition].append(row)
        summary_name = f"{candidate}/summary.json"
        summary = document(summary_name)
        pins = {
            **dict(zip(evaluation.PINS, controls.admission_binding, strict=True)),
            "control_plan_sha256": controls.plan_file_sha256,
            "control_plan_content_sha256": controls.plan_sha256,
            "control_inputs_sha256": controls.sha256,
            "scorer_sha256": evaluation.protocol.SCORER_SHA256,
        }
        aggregates = evaluation._summarize(scores)
        require(summary.get("candidate") == candidate and summary.get("status") == "HOLDOUT_EVALUATED"
                and summary.get("experiment_id") == EXPERIMENT_ID and summary.get("pins") == pins
                and all(summary.get(key) == value for key, value in aggregates.items()),
                "Incomplete or mismatched candidate summary")
        summaries[candidate] = {"conditions": len(conditions), "clips": len(clips),
                                "scored_records": sum(len(rows) for rows in scores.values()),
                                "summary_sha256": copied[summary_name]["sha256"],
                                "model_binding_sha256": copied[binding_name]["sha256"]}
    return summaries


def run_and_capture(argv, work, env, output, out, candidates, clips, plan, controls, *, terminal=None):
    import fsboard_task_evaluate as evaluation

    remote = out / "artifacts"
    remote.mkdir()
    require(len(clips) == 64 and len({clip["id"] for clip in clips}) == 64
            and all(PurePosixPath(clip["id"]).parts == (clip["id"],)
                    and not clip["id"].startswith(".") for clip in clips)
            and len(evaluation.CONDITIONS) == 5, "Unexpected frozen holdout shape")
    expected = {
        f"{candidate}/{clip['id']}/{condition}.json"
        for candidate in candidates for clip in clips for condition in evaluation.CONDITIONS
    }
    expected.update(f"bindings/{candidate}.json" for candidate in candidates)
    expected.update(f"{candidate}/summary.json" for candidate in candidates)
    expected.add("runtime_receipt.json")
    allowed = expected | {"failure.json"}
    child = None
    error = None
    copied = {}
    deadline = None
    strict = False
    summaries = None
    if terminal is None:
        terminal = SimpleNamespace(finalized=False)

    def terminate(signum, frame):
        nonlocal deadline
        if terminal.finalized:
            return
        if deadline is None:
            deadline = time.monotonic() + 15
        if child is not None and child.poll() is None:
            child.terminate()

    previous = signal.signal(signal.SIGTERM, terminate)
    try:
        try:
            child = subprocess.Popen(argv, cwd=work, env=env)
            if deadline is not None and child.poll() is None:
                child.terminate()
            while child.poll() is None:
                if deadline is not None and time.monotonic() >= deadline:
                    child.kill()
                    child.wait()
                    break
                capture_records(output, remote, allowed, copied)
                time.sleep(5)
        except (Exception, KeyboardInterrupt) as failure:
            error = failure
        finally:
            if child is not None and child.poll() is None:
                child.terminate()
                try:
                    child.wait(timeout=15)
                except subprocess.TimeoutExpired:
                    child.kill()
                    child.wait()
            try:
                capture_records(output, remote, allowed, copied, strict=True)
                strict = True
                if child is not None and child.returncode == 0 and deadline is None and error is None:
                    local_json = {
                        path.relative_to(output).as_posix() for path in output.rglob("*.json")
                        if not any(part.startswith(".") for part in path.relative_to(output).parts)
                    }
                    require(set(copied) == expected == local_json, "Incomplete or unexpected evaluation capture")
                    summaries = verify_complete(remote, copied, candidates, clips, plan, controls)
            except (Exception, KeyboardInterrupt) as failure:
                if error is None:
                    error = failure
        rc = None if child is None else child.returncode
        # Close cancellation before publication and carry that boundary through bootstrap cleanup.
        terminal.finalized = True
        cancelled = deadline is not None
        completed = rc == 0 and not cancelled and error is None and strict and summaries is not None
        report = {
            "status": "completed_holdout_evaluation" if completed else "evaluation_incomplete",
            "returncode": rc, "strict_capture_complete": strict, "cancellation_requested": cancelled,
            "orchestration_error_type": None if error is None else type(error).__name__,
            "saved_artifacts": copied,
        }
        if error is not None:
            report["orchestration_error_detail"] = str(error)[:4096]
        if completed:
            report.update(
                candidates=summaries, scored_records=sum(row["scored_records"] for row in summaries.values()),
                pair_success_sha256=PAIR_SUCCESS_SHA, pair_child_exit_sha256=PAIR_EXIT_SHA,
                training_started=False,
            )
        write_json(out / "child-exit.json", report)
        print(report["status"], flush=True)
        return 0 if completed else 130 if cancelled else 1
    finally:
        signal.signal(signal.SIGTERM, previous)

def bootstrap(out):
    terminal = SimpleNamespace(finalized=False)

    def cancel_bootstrap(signum, frame):
        if not terminal.finalized:
            raise KeyboardInterrupt

    signal.signal(signal.SIGTERM, cancel_bootstrap)
    with tempfile.TemporaryDirectory(prefix="sign-to-speak-evaluation-", dir="/tmp") as scratch:
        try:
            return main(out, Path(scratch), terminal=terminal)
        except (Exception, KeyboardInterrupt) as error:
            terminal.finalized = True
            write_json(out / "bootstrap-failure.json", {
                "status": "evaluation_incomplete", "error_type": type(error).__name__, "detail": str(error)[:4096],
            })
            print(f"Evaluation bootstrap stopped ({type(error).__name__})", file=sys.stderr, flush=True)
            return 130 if isinstance(error, KeyboardInterrupt) else 1


if __name__ == "__main__":
    out = JOB / "out/run"
    out.mkdir(mode=0o700)
    raise SystemExit(bootstrap(out))
